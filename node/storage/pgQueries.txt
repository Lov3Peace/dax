-- 1. create table
CREATE TABLE posts (
  id BIGSERIAL PRIMARY KEY,
  user_id BIGINT NOT NULL,
  title TEXT,
  description TEXT,
  created_at TIMESTAMPTZ NOT NULL
);

-- 2. create index for feed queries
CREATE INDEX idx_posts_user_created
ON posts (user_id, created_at DESC);

-- 3. insert data
INSERT INTO posts (user_id, title, description, created_at)
VALUES (42, 'hello', 'first post', now());

-- 4. query (return all columns)
SELECT *
FROM posts
WHERE user_id = 42
ORDER BY created_at DESC
LIMIT 20;

-- 5. paginated query
SELECT *
FROM posts
WHERE user_id = 42
  AND created_at < $1
ORDER BY created_at DESC
LIMIT 20;
